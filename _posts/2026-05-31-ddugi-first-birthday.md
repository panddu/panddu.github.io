// 캡처/발표 모드: 현재 보이는 카드(필터 반영)를 한 장씩 풀스크린으로
(function(){
  var btn=document.getElementById('bqCapBtn'),
      cap=document.getElementById('bqCap'),
      frame=document.getElementById('bqCapFrame'),
      fit=document.getElementById('bqCapFit'),
      cur=document.getElementById('bqCapCur'),
      tot=document.getElementById('bqCapTot'),
      prev=document.getElementById('bqCapPrev'),
      next=document.getElementById('bqCapNext'),
      closeb=document.getElementById('bqCapClose');
  if(!btn||!cap) return;
  var list=[], idx=0;
  function refit(){
    // 폰트 크기만 줄여서 맞춤 → 가로폭은 항상 꽉 채움(텍스트 재줄바꿈)
    fit.style.transform='none';
    var availH=frame.clientHeight*0.80;  // 하단 20%는 영상 자막 공간으로 비움
    var base=window.innerWidth<=600?16:20, min=11;
    var size=base; fit.style.fontSize=size+'px';
    if(fit.scrollHeight>availH){
      size=Math.max(min, size*availH/fit.scrollHeight);
      fit.style.fontSize=size+'px';
      var g=0;
      while(fit.scrollHeight>availH && size>min && g<40){ size-=0.5; fit.style.fontSize=size+'px'; g++; }
    }
  }
  function render(){
    var node=list[idx].cloneNode(true);
    node.classList.remove('is-hidden');
    var ans=node.querySelector('.bq-q__answer'); if(ans) ans.remove();
    // 긴 질문은 캡처용 요약(O열)으로 교체 + '요약' 표기
    var qcap=node.querySelector('.bq-q__qcap');
    if(qcap){
      var qt=node.querySelector('.bq-q__qtext');
      if(qt) qt.innerHTML=qcap.innerHTML;
      var ql=node.querySelector('.bq-q__qlabel');
      if(ql) ql.insertAdjacentHTML('beforeend',' <span class="bq-q__sumtag"><svg viewBox="0 0 24 24" fill="currentColor" aria-hidden="true"><path d="M12 2.2l1.8 5.2 5.2 1.8-5.2 1.8L12 16.2l-1.8-5.2L5 9.2l5.2-1.8z"/><path d="M18.6 13.6l.9 2.5 2.5.9-2.5.9-.9 2.5-.9-2.5-2.5-.9 2.5-.9z"/></svg>AI가 요약한 질문</span>');
      qcap.remove();
    }
    // 메타(세대·성별·MBTI·날짜) + 소개태그(개발자/→에게)를 닉네임 줄로 합쳐 위로 끌어올림
    var hmain=node.querySelector('.bq-q__head-main'),
        hmeta=node.querySelector('.bq-q__head-meta'),
        intro=node.querySelector('.bq-q__tags');
    if(hmain&&hmeta){ while(hmeta.firstChild) hmain.appendChild(hmeta.firstChild); hmeta.remove(); }
    if(hmain&&intro){ while(intro.firstChild) hmain.appendChild(intro.firstChild); intro.remove(); }
    fit.innerHTML=''; fit.appendChild(node);
    cur.textContent=idx+1; tot.textContent=list.length;
    prev.disabled=idx<=0; next.disabled=idx>=list.length-1;
    refit();
  }
  function open(){
    var all=document.querySelectorAll('.bq-list > .bq-q');
    list=Array.prototype.filter.call(all,function(li){return !li.classList.contains('is-hidden');});
    if(!list.length) return;
    idx=0; cap.hidden=false; document.body.style.overflow='hidden'; render();
  }
  function close(){ cap.hidden=true; document.body.style.overflow=''; }
  function go(d){ var n=idx+d; if(n<0||n>=list.length) return; idx=n; render(); }
  btn.addEventListener('click',open);
  closeb.addEventListener('click',close);
  prev.addEventListener('click',function(e){e.stopPropagation();go(-1);});
  next.addEventListener('click',function(e){e.stopPropagation();go(1);});
  cap.addEventListener('click',function(e){
    if(e.target.closest('.bq-cap__stage')||e.target.closest('button')) return;
    var mid=window.innerWidth/2; go(e.clientX<mid?-1:1);
  });
  document.addEventListener('keydown',function(e){
    if(cap.hidden) return;
    if(e.key==='ArrowLeft') go(-1);
    else if(e.key==='ArrowRight') go(1);
    else if(e.key==='Escape') close();
  });
  window.addEventListener('resize',function(){ if(!cap.hidden) refit(); });
})();
</script>
